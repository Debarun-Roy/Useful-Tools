package @PACKAGE@;

import com.password4j.*;
import com.password4j.types.Argon2;
import java.nio.charset.StandardCharsets;
import java.util.UUID;
import java.util.concurrent.Semaphore;

public final class Passwords {
    private static final Argon2Function FUNCTION=Argon2Function.getInstance(19456,2,1,32,Argon2.ID);
    private final Semaphore slots=new Semaphore(2);
    private final String dummy=Password.hash(UUID.randomUUID().toString()).addRandomSalt(16).with(FUNCTION).getResult();
    private final Config config;
    public Passwords(Config config){this.config=config;}
    public void validate(String password) {
        int count=password.codePointCount(0,password.length());
        if(count<config.passwordMin||count>config.passwordMax||password.getBytes(StandardCharsets.UTF_8).length>512)throw new Api.Failure(400,"PASSWORD_POLICY");
    }
    public String hash(String password) {
        validate(password);acquire();try{return Password.hash(password).addRandomSalt(16).with(FUNCTION).getResult();}finally{slots.release();}
    }
    public boolean verify(String password,String hash) {
        validate(password);acquire();try {
            // Only this released policy is accepted. Future policy changes require an explicit rehash migration.
            boolean valid=hash!=null&&hash.matches("\\$argon2id\\$v=19\\$m=19456,t=2,p=1\\$[A-Za-z0-9+/]{22}\\$[A-Za-z0-9+/]{43}");
            boolean matches=Password.check(password,valid?hash:dummy).with(FUNCTION);return valid&&matches;
        }catch(RuntimeException ex){return false;}finally{slots.release();}
    }
    private void acquire(){if(!slots.tryAcquire())throw new Api.Failure(503,"AUTH_BUSY",1);}
}
