"""Explicit supported launcher: one worker, direct TLS, no forwarded-header trust."""
import argparse
import uvicorn
from .config import Config

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--host', default='127.0.0.1')
    parser.add_argument('--port', type=int, default=8443)
    parser.add_argument('--cert')
    parser.add_argument('--key')
    args = parser.parse_args()
    config = Config.load()
    if config.production and (not args.cert or not args.key):
        parser.error('Production requires --cert and --key')
    if not config.production and args.host not in ('127.0.0.1', '::1'):
        parser.error('Development must bind loopback')
    uvicorn.run('@MODULE@.app:create_app', factory=True, host=args.host, port=args.port,
        ssl_certfile=args.cert, ssl_keyfile=args.key, workers=1, proxy_headers=False,
        access_log=False, server_header=False, limit_concurrency=64, backlog=64,
        timeout_keep_alive=5, timeout_graceful_shutdown=10)

if __name__ == '__main__':
    main()
